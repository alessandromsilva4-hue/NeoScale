// O Fechamento possui leitura própria para não depender do módulo de caixa.
async function listarMovimentosFechamento(caixaId=null){
  const base=collection(db,'movimentosCaixa');
  const snap=caixaId
    ? await getDocs(base)
    : await getDocs(base);
  const lista=snap.docs.map(d=>({id:d.id,...d.data()}));
  return caixaId ? lista.filter(m=>String(m.caixaId||'')===String(caixaId)) : lista;
}
function calcularCaixaFechamento(caixa,movs){
  const entrada=movs.filter(m=>['VENDA','SUPRIMENTO'].includes(String(m.tipo||'').toUpperCase())).reduce((a,m)=>a+Number(m.valor||0),0);
  const saida=movs.filter(m=>['SANGRIA','DESPESA'].includes(String(m.tipo||'').toUpperCase())).reduce((a,m)=>a+Number(m.valor||0),0);
  return {entrada,saida,esperado:Number(caixa?.valorInicial||0)+entrada-saida};
}

function formaKey(v){
  const s=String(v||'').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'');
  if(s.includes('dinheiro')) return 'dinheiro';
  if(s.includes('pix')) return 'pix';
  if(s.includes('debito')) return 'debito';
  if(s.includes('credito')) return 'credito';
  return null;
}
function valorInput(id){ return Math.max(0,Number($(id)?.value||0)); }
function setDiff(id,sys,conf){
  const d=conf-sys, el=$(id); if(!el)return; el.textContent=br(d);
  el.className=d===0?'conc-ok':'conc-divergente';
}
function atualizarConciliacao(){
  const map=[['dinheiro','sysDinheiro','maqDinheiro','difDinheiro'],['pix','sysPix','maqPix','difPix'],['debito','sysDebito','maqDebito','difDebito'],['credito','sysCredito','maqCredito','difCredito']];
  let totalS=0,totalC=0;
  for(const [k,si,mi,di] of map){ const sys=valoresSistema[k]||0, conf=valorInput(mi); totalS+=sys; totalC+=conf; $(si).textContent=br(sys); setDiff(di,sys,conf); }
  $('concTotalSistema').textContent=br(totalS); $('concTotalMaquina').textContent=br(totalC);
  const dif=totalC-totalS, status=$('statusConc');
  status.textContent=Math.abs(dif)<0.005?'CAIXA CONCILIADO':'DIVERGÊNCIA ENCONTRADA';
  $('resultadoConc').classList.toggle('conciliado',Math.abs(dif)<0.005);
}

async function obterColecaoSeguro(nome, ms=8000){
  const timeout=new Promise((_,rej)=>setTimeout(()=>rej(new Error('TIMEOUT_'+nome)),ms));
  const snap=await Promise.race([getDocs(collection(db,nome)),timeout]);
  return snap.docs.map(d=>({id:d.id,...d.data()}));
}

 const d=v?.toDate?v.toDate():(v?new Date(v):null);

}

function dentroPeriodo(c, periodo){
 if(periodo==='todos') return true;
 const d=dataObj(c.abertoEm); if(!d)return false;
 const hoje=new Date(); hoje.setHours(0,0,0,0); d.setHours(0,0,0,0);
 if(periodo==='hoje') return d.getTime()===hoje.getTime();
 const dias=Number(periodo); const limite=new Date(hoje); limite.setDate(limite.getDate()-dias+1); return d>=limite;
}
function aplicarFiltrosFechamento(){

  const id=row.dataset.id, c=caixas.find(x=>x.id===id); if(!c)return;
  const operador=(row.dataset.operador||''); const status=row.dataset.status;
  const okBusca=!filtroOperador||operador.includes(filtroOperador.toLowerCase());
  const okPeriodo=dentroPeriodo(c,filtroPeriodo);
  const dif=Math.abs(Number(c._diferenca||0))>=0.005;
  const okStatus=filtroStatus==='todos'||status===filtroStatus||(filtroStatus==='DIVERGENTE'&&status==='FECHADO'&&dif);
  row.style.display=okBusca&&okPeriodo&&okStatus?'':'none';
 }
}

function pedidosDeliveryConcluidos(){
 try{
  const rows=JSON.parse(localStorage.getItem('neoscale_delivery_orders')||'[]');
  return Array.isArray(rows)?rows.filter(o=>String(o.status||o.state||'').toLowerCase().includes('concl') || String(o.status||o.state||'').toLowerCase()==='finalizado'):rows;
 }catch{return []}
}
function chavePlataformaDelivery(v){
 const s=String(v||'').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'');
 if(s.includes('ifood'))return 'ifood'; if(s.includes('99food')||s.includes('99 food'))return 'food99'; if(s.includes('anota'))return 'anota'; if(s.includes('balcao')||s.includes('balcão'))return 'balcao'; return null;
}
function valoresSistemaDelivery(){
 const out={ifood:0,food99:0,anota:0,balcao:0};
 for(const o of pedidosDeliveryConcluidos()){
  const k=chavePlataformaDelivery(o.source||o.platform||o.canal||o.origem);
  if(k)out[k]+=Number(o.total||o.valor||0);
 }
 return out;
}
function atualizarConciliacaoDelivery(){
 const sistema=valoresSistemaDelivery(); let totalS=0,totalC=0;
 for(const [k,,sid,mid,did] of DELIVERY_PLATFORMS){
  const sys=Number(sistema[k]||0), conf=valorInput(mid), dif=conf-sys;
  totalS+=sys; totalC+=conf;
  if($(sid))$(sid).textContent=br(sys);
  setDiff(did,sys,conf);
 }
 if($('deliveryConcTotalSistema'))$('deliveryConcTotalSistema').textContent=br(totalS);
 if($('deliveryConcTotalConferido'))$('deliveryConcTotalConferido').textContent=br(totalC);
 const dif=totalC-totalS;
 if($('deliveryConcDiferenca'))$('deliveryConcDiferenca').textContent=Math.abs(dif)<0.005?'CONCILIADO':br(dif);
 $('deliveryConcResultado')?.classList.toggle('conciliado',Math.abs(dif)<0.005);
}
function preencherCaixaDelivery(){
 const sel=$('caixaConciliacaoDelivery'); if(!sel)return;
 const fechados=caixas.filter(c=>c.status==='FECHADO');
 sel.innerHTML=fechados.map(c=>`<option value="${c.id}">${c.operador||'Operador'} — ${data(c.fechadoEm)} — ${br(c.valorContado)}</option>`).join('');
 if(!sel.options.length){sel.innerHTML='<option value="">Nenhum caixa fechado</option>';}
}
async function prepararConciliacaoDelivery(){
 preencherCaixaDelivery(); atualizarConciliacaoDelivery();
 const sel=$('caixaConciliacaoDelivery');
 if(sel?.value) $('deliveryConcAlerta').innerHTML='<i class="bi bi-check-circle"></i> Caixa selecionado. Confira os valores dos relatórios de cada plataforma do Delivery.';
}
DELIVERY_PLATFORMS.forEach(([, , ,mid])=>$(mid)?.addEventListener('input',atualizarConciliacaoDelivery));
$('deliveryConcObs')?.addEventListener('input',()=>{const el=$('deliveryContadorObs');if(el)el.textContent=String($('deliveryConcObs').value||'').length;});
$('carregarConciliacaoDelivery')?.addEventListener('click',prepararConciliacaoDelivery);
$('salvarConciliacaoDelivery')?.addEventListener('click',async()=>{
 atualizarConciliacaoDelivery();
 const sistema=valoresSistemaDelivery();
 const conferido={}; let totalS=0,totalC=0;
 for(const [k,,sid,mid] of DELIVERY_PLATFORMS){const sys=Number(sistema[k]||0),conf=valorInput(mid);conferido[k]=conf;totalS+=sys;totalC+=conf;}
 const diferenca=totalC-totalS, observacao=String($('deliveryConcObs')?.value||'').trim();
 if(Math.abs(diferenca)>=0.005&&!observacao){alert('Informe uma observação/justificativa para salvar uma divergência no Delivery.');return;}
 try{await addDoc(collection(db,'conciliacoesDelivery'),{sistemaPorPlataforma:sistema,conferidoPorPlataforma:conferido,totalSistema:totalS,totalConferido:totalC,diferenca,status:Math.abs(diferenca)<0.005?'CONCILIADO':'DIVERGENTE',observacao,caixaId:$('caixaConciliacaoDelivery')?.value||null,criadoEm:serverTimestamp()});alert('Conciliação Caixa Delivery salva com sucesso.');}
 catch(e){console.error(e);alert('Não foi possível salvar a conciliação do Delivery.');}
});
preencherCaixaDelivery(); atualizarConciliacaoDelivery();
